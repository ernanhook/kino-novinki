# Основные настройки. Формат: ключ = значение. Строки с # — комментарии.

# Когда присылать подборку: день недели (пн, вт, ср, чт, пт, сб, вс) и час по Риге
digest_day = пт
digest_hour = 18

# Что включать (через запятую): фильмы, сериалы, сезоны
#   фильмы  — фильмы, которые вышли в цифре (стриминг, покупка, онлайн-прокат)
#   сериалы — премьеры новых сериалов
#   сезоны  — новые сезоны уже идущих сериалов
include = фильмы, сериалы, сезоны

# Минимальные оценки. Нужны обе (require = both) или хватит одной (require = any)
min_imdb = 7.5
min_kp = 7.5
require = both

# Минимум голосов, чтобы оценке можно было верить (у свежих релизов голосов мало)
min_imdb_votes = 1000
min_kp_votes = 500

# Сколько дней ждать, пока у новинки наберутся оценки. Фильм, вышедший в цифре
# за эти дни, будет проверяться каждую неделю и придёт, как только наберёт 7.5
lookback_days = 45

# Страна для блока «Где смотреть» (код страны: LV — Латвия, LT, EE, DE, US…)
watch_country = LV

# По каким странам искать даты цифровых релизов фильмов (через запятую)
regions = US, GB, DE

# Жанры, которые не присылать (через запятую, по-русски, как на TMDB):
# документальный, мультфильм, ужасы, реалити-шоу, ток-шоу, новости, детский, мыльная опера
exclude_genres = реалити-шоу, ток-шоу, новости, мыльная опера

# Больше скольких карточек с постерами не присылать (остальное — коротким списком)
max_cards = 20

# Сколько запросов к Кинопоиску можно сделать за один запуск (бесплатно — 200 в сутки)
kp_requests_per_run = 120

# Строка диагностики в конце подборки: auto (если были ошибки), always, never
show_diagnostics = auto

# ===== Рекомендации по запросу =====
# Напишите боту «/kino ужастик» (в личке можно просто «хочу ужастик») — он подберёт новинки
# под ваш вкус из taste.txt (опрос) и ваших 👍/👎 на карточках.
# Сколько фильмов советовать за раз
rec_count = 3
# Насколько свежие новинки (365 = год, 180 = полгода). Если подходящего мало,
# бот сам расширит поиск до двух лет и напишет об этом
rec_days = 365
# Минимальная оценка IMDb и Кинопоиска для советов (ниже, чем для подборки, —
# у ужастиков и комедий 7.5 бывает редко). Если у свежего фильма одной из оценок ещё нет,
# хватает той, что есть
rec_min_rating = 7.0

# ИИ, который понимает запрос и подбирает под вкус (нужен ключ ANTHROPIC_API_KEY;
# без него бот советует просто самое высоко оценённое). Один запрос — доли цента.
model = claude-haiku-5-5
# Лимит расходов на ИИ в месяц, $
monthly_budget_usd = 1.00
